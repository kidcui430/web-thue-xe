-- 1. Bảng quản lý danh sách xe (vehicles)
CREATE TABLE vehicles (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  name TEXT NOT NULL,                  -- Tên xe (VD: Mazda 3, VinFast VF8...)
  type TEXT NOT NULL,                  -- Loại xe (4 chỗ, 7 chỗ, Bán tải...)
  base_price NUMERIC NOT NULL,         -- Giá thuê ngày thường (VNĐ/ngày)
  weekend_price NUMERIC,               -- Giá thuê cuối tuần / Lễ (tuỳ thời điểm)
  status TEXT DEFAULT 'available',     -- Trạng thái nhanh: 'available' (Rảnh), 'rented' (Đang thuê), 'maintenance' (Bảo trì)
  images TEXT[] DEFAULT '{}',          -- Mảng lưu link các hình ảnh của xe
  description TEXT,                    -- Mô tả thông số, tiện ích
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. Bảng lịch đặt xe & Kiểm tra trùng lịch (bookings)
CREATE TABLE bookings (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  vehicle_id BIGINT REFERENCES vehicles(id) ON DELETE CASCADE,
  customer_name TEXT NOT NULL,         -- Tên khách hàng
  customer_phone TEXT NOT NULL,        -- SĐT khách hàng
  start_time TIMESTAMPTZ NOT NULL,     -- Ngày giờ bắt đầu thuê
  end_time TIMESTAMPTZ NOT NULL,       -- Ngày giờ trả xe
  total_price NUMERIC,                 -- Tổng tiền dự kiến
  status TEXT DEFAULT 'pending',       -- 'pending' (Chờ duyệt), 'confirmed' (Đã chốt), 'completed' (Đã trả), 'cancelled' (Huỷ)
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. Bảng để lại thông tin chờ xe trống (waitlist)
CREATE TABLE waitlist (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  vehicle_id BIGINT REFERENCES vehicles(id) ON DELETE CASCADE,
  customer_name TEXT NOT NULL,
  customer_phone TEXT NOT NULL,
  desired_start TIMESTAMPTZ,           -- Thời gian khách muốn thuê
  desired_end TIMESTAMPTZ,
  note TEXT,                           -- Ghi chú thêm
  is_contacted BOOLEAN DEFAULT FALSE,  -- Admin đánh dấu đã liên hệ lại hay chưa
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 4. Bảng quản trị nội dung chung (site_settings)
CREATE TABLE site_settings (
  key TEXT PRIMARY KEY,
  value TEXT NOT NULL
);
